\section{随机变量的分类}

设$f$是概率空间$(X,\mathscr{F},P)$上的$n$维随机变量。由\cref{prop:MeasurableMapping}(4.a)可知$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}),Pf^{-1})$是一个概率空间。因为$Pf^{-1}$是概率测度，所以$Pf^{-1}$是$\sigma$有限测度，而由\cref{prop:LSMeasure}(1)可知L测度$\lambda^n$也是$\sigma$有限测度，根据\cref{theo:LebesgueDecomposition}可知$Pf^{-1}$有如下分解式：
\begin{equation*}
	Pf^{-1}=\mu_1+\mu_s,\quad\mu_1\ll\lambda^n,\quad\mu_s\perp\lambda^n
\end{equation*}
其中$\mu_1$和$\mu_s$是$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}))$上的$\sigma$有限测度，且这一分解是唯一的。由\cref{prop:BorelSigmaField}(1.b)可知$\mathbb{R}^{n}$上的单点集都在$\mathcal{B}(\mathbb{R}^{n})$中，于是记：
\begin{gather*}
	D=\{x\in\mathbb{R}^{n}:\mu_s(\{x\})>0\}
\end{gather*}
则$D$是至多可列集：因为$\mu_s$是$\sigma$有限测度，所以存在满足$\underset{n=1}{\overset{+\infty}{\cup}}A_n=\mathbb{R}^{n}$的$\{A_n\}\subseteq\mathcal{B}(\mathbb{R}^{n})$使得$\mu_s(A_n)<+\infty$对任意的$n$成立。若$D$是不可列集，由\cref{theo:CountableUnionCountable}可知必然有一个$A_n$中有不可列个正测度的单点。取：
\begin{equation*}
	\forall\;k\in\mathbb{N}^+,\;E_k=\left\{x\in A_n:\mu_s(x)\geqslant\frac{1}{k}\right\}
\end{equation*}
设$\mu_s(A_n)=M$，由\cref{prop:Measure}(3)（单调性，次可列可加性）可知$E_k$中最多有$Mk$个元素，根据\cref{theo:CountableUnionCountable}可知$A_n$中正测度的单点数至多可列，矛盾，所以$D$至多可列。\par
因为$D$至多可列，所以$D\in\mathcal{B}(\mathbb{R}^{n})$，于是令：
\begin{equation*}
	\forall\;A\in\mathcal{B}(\mathbb{R}^{n}),\;\mu_2(A)=\mu_s(A\cap D),\;\mu_3(A)=\mu_s(A)-\mu_2(A)
\end{equation*}
因为$\mu_s$是$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}))$上的测度，由定义可验证得到$\mu_2$是$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}))$上的测度，由\cref{prop:Measure}(3)（次有限可加性）还可得到$\mu_2$是有限的，根据\cref{prop:Measure}(2)(3)（次有限可加性）可知$\mu_3$是有限的。\par
综上，$Pf^{-1}$有分解：
\begin{equation*}
	Pf^{-1}=\mu_1+\mu_2+\mu_3
\end{equation*}\par
若对任意的$A\in\mathcal{B}(\mathbb{R}^{n})$有$\mu_2(A)=\mu_3(A)=0$，称$f$为\gls{ContinuousRandomVector}。因为$\mu_1\ll\lambda^n$，所以此时有$Pf^{-1}\ll\lambda^n$，因为$\lambda^n$是$\sigma$有限测度，$Pf^{-1}$是概率测度，由\cref{theo:RadonNikodym}可知存在$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}),\lambda^n)$上在a.e.意义下唯一的可测函数$p$满足对任意的$A\in\mathcal{B}(\mathbb{R}^{n})$有：
\begin{equation*}
	Pf^{-1}(A)=\int_{A}p(x)\dif\lambda^n
\end{equation*}
由\cref{prop:MeasurableIntegral}(9)还可证明得到$p\geqslant0\;$a.e.于$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}),\lambda^n)$，称$p$为$f$的\gls{P.D.F}。在没有明确随机变量的场合，也将概率测度$P$对L测度的Radon-Nikodym导数称为概率密度函数。
\begin{note}
	在证明随机向量$f$为连续型随机向量时，只需证明存在满足上述条件的$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}))$上的非负可测函数$p$。\par
	由\cref{prop:MeasurableIntegral}(1)可知此时$Pf^{-1}\ll\lambda^n$，所以$Pf^{-1}=Pf^{-1}+0$即为$Pf^{-1}$关于$\lambda^n$的Lebesgue分解。由\cref{theo:LebesgueDecomposition}中Lebesgue分解的唯一性可知$\mu_2=\mu_3=0$，所以$f$为连续型随机向量，此时$p$即为$f$的概率密度函数。
\end{note}
若对任意的$A\in\mathcal{B}(\mathbb{R}^{n})$有$\mu_1(A)=\mu_3(A)=0$，称$f$为\gls{DiscreteRandomVector}\footnote{从这里开始我们可能会混用初等概率论的记号，即用$P(f=x_n)$来表示$P(\{f=x_n\})$。}。记点列$D=\{x_n\},\;P(f=x_n)=Pf^{-1}(\{x_n\}),\;n\in\mathbb{N}^+$，则：
\begin{equation*}
	p(x)\coloneq
	\begin{cases}
		P(f=x),&x\in D \\
		0,&x\notin D
	\end{cases}
\end{equation*}
完全确定了$f$的概率分布，称$p$\footnote{对于概率质量函数$p$，通常只写出其非零部分，未列出的点均约定其概率质量为$0$。}为$f$的\gls{P.M.F}（把$p$非零的部分写成表格的形式后称其为概率分布列，或简称为分布列）。在没有明确随机变量的场合，也将满足条件的$\mu_2$称为概率质量函数。

\begin{note}
	在证明随机向量$f$为离散型随机向量时，只需证明存在至多可数集$D\subseteq\mathbb{R}^{n}$使得$P(f\in D)=1$。\par
	由\cref{prop:Measure}(1)可知$Pf^{-1}(D^c)=0$，因为：
	\begin{equation*}
		Pf^{-1}=\mu_1+\mu_2+\mu_3
	\end{equation*}
	且$\mu_1,\mu_2,\mu_3$均为测度，所以：
	\begin{equation*}
		\mu_1(D^c)=\mu_2(D^c)=\mu_3(D^c)=0
	\end{equation*}\par
	因为$D$至多可数，由\cref{prop:LSMeasure}(6)可知$\lambda^n(D)=0$。又因为$\mu_1\ll\lambda^n$，所以$\mu_1(D)=0$。结合$\mu_1(D^c)=0$，根据\cref{prop:Measure}(1)(3)（单调性）可得$\mu_1=0$。\par
	由$\mu_3$的构造可知对于任意的$x\in\mathbb{R}^{n}$均有$\mu_3(\{x\})=0$。根据\cref{prop:Measure}(3)（次可列可加性）可得$\mu_3(D)=0$，与$\mu_1$同理可知$\mu_3=0$。\par
	综上，$f$为离散型随机向量。
\end{note}
若对任意的$A\in\mathcal{B}$有$\mu_1(A)=\mu_2(A)=0$，称$f$为\gls{SingularRandomVector}。\par
统称概率质量函数与概率密度函数为\gls{ProbabilityFunction}。
\begin{theorem}
	任何随机向量都是离散型、连续型、奇异型随机向量的混合。
\end{theorem}
\begin{definition}
	称一族概率测度$\mathscr{P}$为\gls{DistributionFamily}。
\end{definition}
\begin{property}\label{prop:ContinuousRV}
	设$f,g$分别为概率空间$(X,\mathscr{F},P)$上的$m$维和$n$维连续型随机向量，则：
	\begin{enumerate}
		\item 对于任意的$a,b\in\mathbb{R}^{n}$，$P(f\leqslant a)=P(f<a),\;P(f\geqslant a)=P(f>a),\;P(a<f<b)=P(a\leqslant f<b)=P(a<f\leqslant b)=P(a\leqslant f\leqslant b)$；
		\item $(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}))$上的可测函数$p$是某个$n$维连续型随机向量的概率密度函数的充要条件为：
		\begin{enumerate}
			\item $p(x)\geqslant0\;$a.e.于$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}),\lambda^n)$；
			\item $\int_{\mathbb{R}^{n}}p(x)\dif\lambda^n=1$。
		\end{enumerate}
	\end{enumerate}
\end{property}
\begin{proof}
	(1)由\cref{prop:MeasurableFunction}(1)(2)、\cref{prop:Measure}(1)和连续型随机向量的定义即可得到。\par
	(2)必要性由定义即可得到，下证明充分性。\par
	令$\widetilde{p}(x)=\max\{p(x),0\}$，由\cref{prop:MeasurableFunction}(6)可知$\widetilde{p}$为非负可测函数，并且$\widetilde{p}=p\;$a.e.于$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}),\lambda^n)$，从而：
	\begin{equation*}
		\int_{\mathbb{R}^{n}}\widetilde{p}(x)\dif\lambda^n=1
	\end{equation*}
	对于任意的$A\in\mathcal{B}(\mathbb{R}^{n})$，定义：
	\begin{equation*}
		\mu(A)=\int_{A}\widetilde{p}(x)\dif\lambda^n
	\end{equation*}
	由\cref{prop:NonnegativeMeasurableIntegral}(2)和\cref{theo:MeasurableCountableIntegral}可知$\mu$是$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}))$上的概率测度，因此$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}),\mu)$为概率空间。\par
	定义：
	\begin{equation*}
		f:\mathbb{R}^{n}\to\mathbb{R}^{n},\quad f(x)=x
	\end{equation*}
	由\cref{prop:BorelSigmaField}(1.b)和\cref{theo:ProductMeasurable}可知$f$为概率空间$(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}),\mu)$上的$n$维随机向量。对于任意的$A\in\mathcal{B}(\mathbb{R}^{n})$有：
	\begin{equation*}
		f^{-1}(A)=A
	\end{equation*}
	于是由\cref{prop:MeasurableIntegral}(8)可知：
	\begin{equation*}
		\mu(f\in A)=\mu(A)=\int_{A}\widetilde{p}(x)\dif\lambda^n=\int_{A}p(x)\dif\lambda^n
	\end{equation*}
	因此$f$为连续型随机向量，并且$p$为$f$的概率密度函数，故充分性成立。
\end{proof}
\begin{property}\label{prop:DiscreteRV}
	设$f,g$分别为概率空间$(X,\mathscr{F},P)$上的$m$维和$n$维离散型随机向量，则：
	\begin{enumerate}
		\item 记$f$的概率质量函数为$p_f$，则对于任意的$A\in\mathcal{B}(\mathbb{R}^m)$有：
		\begin{equation*}
			P(f\in A)=\sum_{\substack{x\in A\\p_f(x)>0}}p_f(x)
		\end{equation*}
		\item 记$(f,g)$的联合概率质量函数为$p_{f,g}$，定义：
		\begin{equation*}
			p_f(x)=\sum_{\substack{y\in\mathbb{R}^n\\p_{f,g}(x,y)>0}}p_{f,g}(x,y),\qquad
			p_g(y)=\sum_{\substack{x\in\mathbb{R}^m\\p_{f,g}(x,y)>0}}p_{f,g}(x,y)
		\end{equation*}
		则$p_f$和$p_g$分别为$f$和$g$的概率质量函数；
		\item $(\mathbb{R}^{n},\mathcal{B}(\mathbb{R}^{n}))$上的可测函数$p$是某个$n$维离散型随机向量的概率质量函数的充要条件为：
		\begin{enumerate}
			\item 对于任意的$x\in\mathbb{R}^n$有$p(x)\geqslant0$；
			\item $\sum\limits_{\substack{x\in\mathbb{R}^n\\p(x)>0}}p(x)=1$。
		\end{enumerate}
	\end{enumerate}
\end{property}
\begin{proof}
	(1)由离散型随机变量的定义和测度的可列可加性即可得出。\par
	(2)对于任意的$x\in\mathbb{R}^m$，记：
	\begin{equation*}
		D_{f,g}=\left\{(u,v)\in\mathbb{R}^{m+n}:p_{f,g}(u,v)>0\right\}
	\end{equation*}
	由于$(f,g)$为离散型随机向量，所以$P((f,g)\in D_{f,g})=1$，根据\cref{prop:Measure}(1)可知$P((f,g)\notin D_{f,g})=0$。由\cref{prop:SetOperation}(4)和\cref{prop:Measure}(1)(3)（单调性）可知：
	\begin{align*}
		P(f=x)&=P\Big(\{f=x\}\cap\{(f,g)\in D_{f,g}\}\Big)+P\Big(\{f=x\}\cap\{(f,g)\notin D_{f,g}\}\Big) \\
		&=P\Big(\{f=x\}\cap\{(f,g)\in D_{f,g}\}\Big)
	\end{align*}
	因为：
	\begin{equation*}
		\{f=x\}\cap\{(f,g)\in D_{f,g}\}=\bigcup_{\substack{y\in\mathbb{R}^n\\p_{f,g}(x,y)>0}}\{f=x,g=y\}
	\end{equation*}
	且上式右侧为至多可数个两两不交事件之并，所以由测度的可列可加性可得：
	\begin{equation*}
		P(f=x)=\sum_{\substack{y\in\mathbb{R}^n\\p_{f,g}(x,y)>0}}P(f=x,g=y)=\sum_{\substack{y\in\mathbb{R}^n\\p_{f,g}(x,y)>0}}p_{f,g}(x,y)=p_f(x)
	\end{equation*}\par
	(3)必要性由定义和(1)即可得到。下证明充分性。\par
	记：
	\begin{equation*}
		D=\left\{x\in\mathbb{R}^n:p(x)>0\right\}
	\end{equation*}
	由条件可知$D$为至多可数集。对于任意的$A\subseteq D$定义：
	\begin{equation*}
		P(A)=\sum_{\substack{x\in A\\p(x)>0}}p(x)
	\end{equation*}
	并规定$P(\varnothing=0)$，于是$(D,\mathscr{P}(D),P)$构成一个概率空间。\par
	定义：
	\begin{equation*}
		f:D\to\mathbb{R}^{n},\quad f(x)=x
	\end{equation*}
	由于$\mathscr{P}(D)$是$D$的所有子集构成的集族，所以根据定义可知$f$为可测映射。由测度的可列可加性和条件可知$f$为离散型随机向量。对于任意的$x\in D$有：
	\begin{equation*}
		P(f=x)=P(\{x\})=p(x)
	\end{equation*}
	对于任意的$x\notin D$有$p(x)=0$，同时：
	\begin{equation*}
		P(f=x)=P(\varnothing)=0=p(x)
	\end{equation*}
	因此对于任意的$x\in\mathbb{R}^n$都有$P(f=x)=p(x)$，即$p$为离散型随机向量$f$的概率质量函数。
\end{proof}
