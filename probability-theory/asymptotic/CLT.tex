\section{中心极限定理}

\begin{theorem}\label{theo:CLT}
	设$\{f_n\}_{n=1}^{\infty}$为概率空间$(X,\mathscr{F},P)$上一列独立同分布的$m$维随机向量，根据\cref{prop:Norm}(2)、\cref{prop:MeasurableFunction}(11)、\cref{prop:MeasurableMapping}(2)和\cref{prop:MeasurableFunction}(5.b)可知$||f_1||^2$是$(X,\mathscr{F},P)$上的随机变量。若$\operatorname{E}(||f_1||^2)<+\infty$，记：
	\begin{equation*}
		\mu=\operatorname{E}(f_1),\quad\Sigma=\operatorname{Cov}(f_1)
	\end{equation*}
	则：
	\begin{equation*}
		\sqrt{n}\left(\frac{1}{n}\sum_{i=1}^{n}f_i-\mu\right)\overset{d}{\longrightarrow}\operatorname{N}_m(0,\Sigma)
	\end{equation*}
\end{theorem}
\begin{definition}
	设$\{f_n\}_{n=1}^{\infty}$为概率空间$(X,\mathscr{F},P)$一列独立同分布的$m$维随机向量。若$\operatorname{E}(||f_1||^2)<+\infty$。记：
	\begin{equation*}
		\mu=\operatorname{E}(f_1),\quad\Sigma=\operatorname{Cov}(f_1)
	\end{equation*}
	根据\cref{theo:CLT}有：
	\begin{equation*}
		\sqrt{n}\left(\frac{1}{n}\sum_{i=1}^{n}f_i-\mu\right)\overset{d}{\longrightarrow}\operatorname{N}_m(0,\Sigma)
	\end{equation*}
	称$\operatorname{N}_d\left(\mu,\dfrac{\Sigma}{n}\right)$为$\dfrac{1}{n}\sum\limits_{i=1}^{n}f_i$的\gls{AsymptoticApproximateDistribution}，记作：
	\begin{equation*}
		\frac{1}{n}\sum_{i=1}^{n}f_i\overset{a}{\longrightarrow}\operatorname{N}_m\left(\mu,\frac{\Sigma}{n}\right)
	\end{equation*}
\end{definition}
