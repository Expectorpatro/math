\section{线性方程组}

\subsection{基本对象与记号}
\begin{definition}
	设$K$为数域，$a_1,a_2,\dots,a_n,b\in K$，$x_1,x_2,\dots,x_n$为未知量。形如
	\begin{equation*}
		a_1x_1+a_2x_2+\cdots+a_nx_n=b
	\end{equation*}
	的方程称为\gls{LinearEquation}。由$m$个线性方程
	\begin{equation*}
		\sum_{j=1}^{n}a_{ij}x_j=b_i,\quad i=1,2,\dots,m
	\end{equation*}
	组成的方程组称为$n$元\gls{SLE}。令：
	\begin{equation*}
		A=(a_{ij})\in M_{m\times n}(K),\;x=(\seq{x}{n})^{\top}\in K^n,\;b=(\seq{b}{m})^{\top}\in K^m
	\end{equation*}
	则上述方程组可以简写为$Ax=b$。
\end{definition}
\begin{definition}
	线性方程组$Ax=b$的解集定义为：
	\begin{equation*}
		\mathcal{S}(A,b)\coloneq\{x\in K^n:Ax=b\}
	\end{equation*}
	若$\mathcal{S}(A,b)\ne\varnothing$，则称方程组是\textbf{相容的}；否则称它是\textbf{不相容的}。当$b=\mathbf{0}$时，称$Ax=\mathbf{0}$为\gls{HomogeneousSystemOfLinearEquations}；当$b\ne\mathbf{0}$时，称$Ax=b$为\gls{NonhomogeneousSystemOfLinearEquations}。
\end{definition}
\begin{definition}
	矩阵$A\in M_{m\times n}(K)$可以诱导线性映射：
	\begin{equation*}
		\mathcal{T}_A:K^n\to K^m,\quad x\to Ax
	\end{equation*}
	定义$A$的核与象分别为：
	\begin{gather*}
		\operatorname{Ker}(A)\coloneq\operatorname{Ker}\mathcal{T}_A=\{x\in K^n:Ax=\mathbf{0}\}=\mathcal{S}(A,\mathbf{0}) \\
		\operatorname{Im}(A)\coloneq\operatorname{Im}\mathcal{T}_A=\{Ax:x\in K^n\}=\mathcal{M}(A)
	\end{gather*}
\end{definition}
\begin{definition}
	给定线性方程组$Ax=b$，将系数矩阵$A$与常数列$b$并列得到的矩阵
	\begin{equation*}
		[A|b]\in M_{m\times(n+1)}(K)
	\end{equation*}
	称为该方程组的\gls{AugmentedMatrix}。
\end{definition}

\subsection{行化简与解的判别}
\begin{property}\label{prop:SLEElementaryRowOperation}
	设矩阵$[B|c]$由增广矩阵$[A|b]$经过有限次初等行变换得到，则：
	\begin{equation*}
		\mathcal{S}(A,b)=\mathcal{S}(B,c) 
	\end{equation*}
	即对增广矩阵作初等行变换不改变线性方程组的解集。
\end{property}
\begin{proof}
	由\cref{prop:ElementaryMatrix}(2)可知每次初等行变换都等价于在矩阵左侧乘一个初等矩阵。将有限次变换合并，根据\cref{prop:InvertibleMatrix}(14)可知存在可逆矩阵$P$使得：
	\begin{equation*}
		[B|c]=P[A|b]
	\end{equation*}
	即$B=PA,\;c=Pb$，于是对任意的$x\in K^n$有：
	\begin{equation*}
		Ax=b\iff P(Ax-b)=\mathbf{0}\iff Bx=c\qedhere
	\end{equation*}
\end{proof}
\begin{definition}
	将增广矩阵$[A|b]$化为简化行阶梯形矩阵$[R|c]$。若$R$的主元列依次为第$p_1,p_2,\dots,p_r$列，则称$x_{p_1},x_{p_2},\dots,x_{p_r}$为\gls{PivotVariable}，其余未知量为\gls{FreeVariable}。
\end{definition}
\begin{theorem}\label{theo:SolutionOfSLE1}
	设$[R|c]$是数域$K$上线性方程组$Ax=b$的增广矩阵的简化行阶梯形矩阵，$r$为$R$的主元个数。方程组的解恰有下列三种互斥情形：
	\begin{enumerate}
		\item \textbf{无解：}$[R|c]$中存在形如
		\begin{equation*}
			[0\ \ 0\ \ \cdots\ \ 0\mid d],\quad d\ne0
		\end{equation*}
		的行；
		\item \textbf{唯一解：}不存在上述矛盾行且$r=n$；
		\item \textbf{无穷多解：}不存在上述矛盾行且$r<n$。
	\end{enumerate}
	特别地，齐次线性方程组$Ax=\mathbf{0}$有非零解当且仅当$r<n$；若$m<n$，则它一定有非零解。
\end{theorem}
\begin{proof}
	由\cref{prop:SLEElementaryRowOperation}，只需讨论简化后的方程组$Rx=c$。若出现$0=d$且$d\ne0$，则任何$x$都不能满足该行方程，故方程组无解。

	以下设不存在矛盾行。记主元列为$p_1,p_2,\dots,p_r$，自由未知量的下标为$q_1,q_2,\dots,q_{n-r}$。简化行阶梯形方程组的第$i$个非零方程可以写为
	\begin{equation*}
		x_{p_i}=c_i-\sum_{j=1}^{n-r}R(i;q_j)x_{q_j},
		\quad i=1,2,\dots,r.
	\end{equation*}
	因此，一旦给定全部自由未知量，主变量便被唯一确定。当$r=n$时没有自由未知量，解唯一；当$r<n$时至少有一个自由未知量，而数域$K$含有无穷多个元素，故可得到无穷多个不同的解。

	齐次线性方程组总是相容的，并且其零解对应于所有自由未知量均取$0$。因此，它有非零解当且仅当存在自由未知量，即$r<n$。又$r\leqslant m$，所以$m<n$蕴含$r<n$。
\end{proof}

\subsection{相容性与秩判别}

\begin{theorem}\label{theo:SolutionOfSLE2}
	设$A=(\seq{\alpha}{n})\in M_{m\times n}(K)$，其中$\alpha_1,\alpha_2,\dots,\alpha_n$为$A$的列向量，$b\in K^m$。则：
	\begin{enumerate}
		\item 方程组$Ax=b$相容当且仅当
		\begin{equation*}
			b\in\mathcal{M}(A)=<\seq{\alpha}{n}>;
		\end{equation*}
		\item 方程组$Ax=b$相容当且仅当
		\begin{equation*}
			\operatorname{rank}(A)=\operatorname{rank}([A|b]).
		\end{equation*}
		在上述相容条件成立时：
		\begin{enumerate}
			\item 方程组有唯一解当且仅当$\operatorname{rank}(A)=n$；
			\item 方程组有无穷多个解当且仅当$\operatorname{rank}(A)<n$；
		\end{enumerate}
		\item 矩阵$A$的核与象满足
		\begin{equation*}
			\dim\operatorname{Ker}(A)=n-\operatorname{rank}(A),
			\quad
			\dim\operatorname{Im}(A)=\operatorname{rank}(A).
		\end{equation*}
	\end{enumerate}
\end{theorem}
\begin{proof}
	(1)由矩阵乘法的定义，
	\begin{equation*}
		Ax=\sum_{i=1}^{n}x_i\alpha_i.
	\end{equation*}
	因此，存在$x\in K^n$使$Ax=b$，当且仅当$b$是$\alpha_1,\alpha_2,\dots,\alpha_n$的线性组合。

	(2)显然有$\mathcal{M}(A)\subseteq\mathcal{M}([A|b])$。结合(1)可得
	\begin{equation*}
		Ax=b\text{相容}
		\iff\mathcal{M}(A)=\mathcal{M}([A|b]).
	\end{equation*}
	由\cref{prop:SpanSubspace}(3)与\cref{prop:Subspace}(4)，上述两个有限维子空间相等，当且仅当它们的维数相等，也即
	\begin{equation*}
		\operatorname{rank}(A)=\operatorname{rank}([A|b]).
	\end{equation*}
	在相容的前提下，系数矩阵化成阶梯形矩阵后的非零行数等于$\operatorname{rank}(A)$。于是(2.1)与(2.2)由\cref{theo:SolutionOfSLE1}立即得到。

	(3)考虑$A$诱导的线性映射$\mathcal{T}_A:K^n\longrightarrow K^m$。由定义，
	\begin{equation*}
		\operatorname{Ker}\mathcal{T}_A=\operatorname{Ker}(A),
		\quad
		\operatorname{Im}\mathcal{T}_A=\operatorname{Im}(A)=\mathcal{M}(A).
	\end{equation*}
	又由\cref{prop:SpanSubspace}(3)，
	\begin{equation*}
		\dim\operatorname{Im}(A)=\dim\mathcal{M}(A)=\operatorname{rank}(A).
	\end{equation*}
	对$\mathcal{T}_A$应用\cref{prop:LinearMapping}(10)，可得
	\begin{equation*}
		n=\dim K^n
		=\dim\operatorname{Ker}(A)+\dim\operatorname{Im}(A),
	\end{equation*}
	结论成立。
\end{proof}

\subsection{解集的结构}

\subsubsection{齐次线性方程组解的结构}

\begin{property}\label{prop:HomogeneousSLESolution}
	设$A\in M_{m\times n}(K)$，齐次线性方程组$Ax=\mathbf{0}$的解具有如下性质：
	\begin{enumerate}
		\item 若$\alpha,\beta$是解，则对任意的$c_1,c_2\in K$，$c_1\alpha+c_2\beta$也是解；
		\item 解集$W=\mathcal{S}(A,\mathbf{0})=\operatorname{Ker}(A)$是$K^n$的子空间；
		\item 解空间$W$满足
		\begin{equation*}
			\dim W=n-\operatorname{rank}(A).
		\end{equation*}
	\end{enumerate}
\end{property}
\begin{proof}
	(1)若$A\alpha=A\beta=\mathbf{0}$，则
	\begin{equation*}
		A(c_1\alpha+c_2\beta)
		=c_1A\alpha+c_2A\beta
		=\mathbf{0}.
	\end{equation*}
	(2)也可以由(1)直接验证；从线性映射的观点看，它就是\cref{prop:LinearMapping}(6)关于$\operatorname{Ker}\mathcal{T}_A$的结论。\par
	(3)由\cref{theo:SolutionOfSLE2}(3)立即得到。
\end{proof}

\begin{definition}
	齐次线性方程组$Ax=\mathbf{0}$的解空间$W=\operatorname{Ker}(A)$的任意一组基称为该方程组的\gls{FundamentalSolutionSet}。由\cref{prop:HomogeneousSLESolution}(3)，基础解系所含向量的个数为$n-\operatorname{rank}(A)$。
\end{definition}

\begin{method}
	设$R$是$A$的简化行阶梯形矩阵，$\operatorname{rank}(A)=r<n$。记$R$的主元列为
	\begin{equation*}
		p_1<p_2<\cdots<p_r,
	\end{equation*}
	自由未知量所在列为
	\begin{equation*}
		q_1<q_2<\cdots<q_{n-r}.
	\end{equation*}
	对$j=1,2,\dots,n-r$，定义向量$\eta_j\in K^n$的坐标为
	\begin{align*}
		(\eta_j)_{q_\ell}
		&=
		\begin{cases}
			1,&\ell=j,\\
			0,&\ell\ne j,
		\end{cases}
		\quad \ell=1,2,\dots,n-r,\\
		(\eta_j)_{p_i}
		&=-R(i;q_j),
		\quad i=1,2,\dots,r.
	\end{align*}
	由$R$的结构可知$R\eta_j=\mathbf{0}$。再由\cref{prop:SLEElementaryRowOperation}，$A\eta_j=\mathbf{0}$。这些向量在自由未知量对应的坐标上组成单位矩阵，因而线性无关；任意齐次解又都由其自由未知量的取值唯一确定，因而
	\begin{equation*}
		\{\eta_1,\eta_2,\dots,\eta_{n-r}\}
	\end{equation*}
	是一个基础解系，齐次方程组的通解为
	\begin{equation*}
		x=\sum_{j=1}^{n-r}c_j\eta_j,\quad c_j\in K.
	\end{equation*}
	当$r=n$时，$\operatorname{Ker}(A)=\{\mathbf{0}\}$，其基础解系为空集。
\end{method}

\subsubsection{非齐次线性方程组解的结构}

\begin{property}\label{prop:InhomogeneousSLESolution}
	设非齐次线性方程组$Ax=b$相容，$W=\operatorname{Ker}(A)$，则其解具有如下性质：
	\begin{enumerate}
		\item 若$\alpha,\beta$都是$Ax=b$的解，则$\alpha-\beta\in W$；
		\item 若$\alpha$是$Ax=b$的解，则对任意的$\beta\in W$，$\alpha+\beta$也是$Ax=b$的解；
		\item 若$\alpha$是$Ax=b$的任意一个解，则方程组的解集$U$可以表示为
		\begin{equation*}
			U=\alpha+W
			\coloneq\{\alpha+\beta:\beta\in W\};
		\end{equation*}
		\item 方程组$Ax=b$的解唯一，当且仅当
		\begin{equation*}
			W=\{\mathbf{0}\},
		\end{equation*}
		也当且仅当$\operatorname{rank}(A)=n$。
	\end{enumerate}
\end{property}
\begin{proof}
	(1)因为$A\alpha=A\beta=b$，所以
	\begin{equation*}
		A(\alpha-\beta)=A\alpha-A\beta=\mathbf{0},
	\end{equation*}
	即$\alpha-\beta\in W$。\par
	(2)若$\beta\in W$，则
	\begin{equation*}
		A(\alpha+\beta)=A\alpha+A\beta=b.
	\end{equation*}
	(3)由(2)可知$\alpha+W\subseteq U$；反之，任取$x\in U$，由(1)有$x-\alpha\in W$，故$x\in\alpha+W$。\par
	(4)由(3)，解的唯一性等价于$W=\{\mathbf{0}\}$。再由\cref{prop:HomogeneousSLESolution}(3)，这等价于$\operatorname{rank}(A)=n$。
\end{proof}

\begin{corollary}\label{cor:SLEGeneralSolution}
	设$Ax=b$相容，$x_p$为它的一个特解，$\{\eta_1,\eta_2,\dots,\eta_{n-r}\}$为$Ax=\mathbf{0}$的一个基础解系，其中$r=\operatorname{rank}(A)$。则$Ax=b$的通解为
	\begin{equation*}
		x=x_p+\sum_{j=1}^{n-r}c_j\eta_j,\quad c_j\in K.
	\end{equation*}
\end{corollary}
\begin{proof}
	由\cref{prop:InhomogeneousSLESolution}(3)与基础解系的定义立即得到。
\end{proof}

\subsection{Gauss消元法}

Gauss消元法将\cref{prop:SLEElementaryRowOperation}具体化：先通过前向消元把增广矩阵化为行阶梯形矩阵，再检查相容性；若方程组相容，则继续化为简化行阶梯形矩阵，并据此构造特解与基础解系。理论上，这一过程适用于任意数域$K$。下面给出实数域上采用列主元选取的数值形式。

\begin{algorithm}[H]
	\caption{Gauss前向消元与相容性检查}
	\label{alg:gauss}
	\begin{algorithmic}[1]
		\Require $C=[A|b]=(c_{ij})\in M_{m\times(n+1)}(\mathbb{R}^{})$，容差$\varepsilon>0$
		\Ensure 无解，或行阶梯形增广矩阵$C$及有序主元列指标表$\mathcal{P}$
		\State $s\gets1,\;\mathcal{P}\gets()$ \Comment{$s$为当前主元行}
		\For{$j=1$ \textbf{to} $n$}
			\If{$s>m$}
				\State \textbf{break}
			\EndIf
			\State 取$p\in\{s,s+1,\dots,m\}$使$|c_{pj}|=\max_{s\leqslant i\leqslant m}|c_{ij}|$
			\If{$|c_{pj}|\leqslant\varepsilon$}
				\State \textbf{continue} \Comment{第$j$列没有可用主元}
			\EndIf
			\State 交换$C$的第$p$行与第$s$行
			\For{$i=s+1$ \textbf{to} $m$}
				\State $\ell\gets c_{ij}/c_{sj}$
				\State 将$C$的第$i$行减去第$s$行的$\ell$倍，并令$c_{ij}\gets0$
			\EndFor
			\State 将$j$添加到$\mathcal{P}$的末尾，$s\gets s+1$
		\EndFor
		\State $r\gets|\mathcal{P}|$
		\For{$i=r+1$ \textbf{to} $m$}
			\If{$|c_{i,n+1}|>\varepsilon$且$|c_{ij}|\leqslant\varepsilon\;(j=1,2,\dots,n)$}
				\State \Return 方程组无解
			\EndIf
		\EndFor
		\State \Return $C$与$\mathcal{P}$
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
	\caption{Gauss--Jordan回代与通解构造}
	\label{alg:gaussGeneralSolution}
	\begin{algorithmic}[1]
		\Require 相容行阶梯形矩阵$C$与$\mathcal{P}=(p_1,p_2,\dots,p_r)$
		\Ensure 特解$x_p$、$\operatorname{Ker}(A)$的一组基及$Ax=b$的通解
		\For{$i=r$ \textbf{downto} $1$}
			\State $j\gets p_i$，将$C$的第$i$行除以$c_{ij}$
			\For{$k=1$ \textbf{to} $i-1$}
				\State $\ell\gets c_{kj}$，将$C$的第$k$行减去第$i$行的$\ell$倍
			\EndFor
		\EndFor
		\State 按升序记非主元列指标为$\mathcal{F}=(f_1,f_2,\dots,f_q)$，其中$q=n-r$
		\State $x_p\gets\mathbf{0}\in\mathbb{R}^n$
		\For{$i=1$ \textbf{to} $r$}
			\State $(x_p)_{p_i}\gets c_{i,n+1}$
		\EndFor
		\For{$j=1$ \textbf{to} $q$}
			\State $\eta_j\gets\mathbf{0}\in\mathbb{R}^n,\;(\eta_j)_{f_j}\gets1$
			\For{$i=1$ \textbf{to} $r$}
				\State $(\eta_j)_{p_i}\gets-c_{i,f_j}$
			\EndFor
		\EndFor
		\State \Return $x=x_p+\sum_{j=1}^{q}t_j\eta_j,\quad t_j\in\mathbb{R}^{}$
	\end{algorithmic}
\end{algorithm}
