\section{微分}
微分学研究映射在局部范围内随自变量变化而产生的变化规律，并以线性近似刻画这种变化。
\subsection{Fr\'echet微分}
\begin{definition}
	设$X,Y$为实赋范线性空间，$E\subseteq X$是开集，$x_0\in E$，映射$T:E\to Y$。若存在有界线性算子$A\in\mathcal{L}(X,Y)$满足：
	\begin{equation*}
		\lim_{||h||_X\to0}\frac{||T(x_0+h)-Tx_0-Ah||_Y}{||h||_X}=0
	\end{equation*}
	则称映射$T$在点$x_0$处Fr\'echet \gls{Differentiable}，称$A$是$T$在点$x_0$处的Fr\'echet \gls{Differential}，记作$\operatorname{D}Tx_0$。若$T$在$E$中的每一个点处都Fr\'echet可微，则称$T$在$E$上Fr\'echet可微。若$T$在$E$上Fr\'echet可微，映射$\operatorname{D}T:E\to\mathcal{L}(X,Y)$在点$x_0$处Fr\'echet可微，则称$T$在点$x_0$处二阶Fr\'echet可微，并将$\operatorname{D}T$在点$x_0$处的Fr\'echet微分记作$\operatorname{D}^2Tx_0$，更高阶的情况可依此类推。
\end{definition}
\begin{note}
	Fr\'echet微分的本质是描述映射在某一点附近的局部变化。若映射$T:E\to Y$在$x\in E$处Fr\'echet可微，则$\operatorname{D}Tx$给出了当自变量发生微小变化$h\in X$时，$T$的一阶变化：
	\begin{equation*}
		T(x+h)-T(x)=\operatorname{D}Tx(h)+o(||h||_X)
	\end{equation*}
	由于$\operatorname{D}Tx$需要将自变量的增量$h\in X$映射为函数值的增量$\operatorname{D}Tx(h)\in Y$，因此自然的有$\operatorname{D}Tx\in\mathcal{L}(X,Y)$。\par
	当$x$在$E$中变化时，不同的点对应不同的一阶微分算子。因此，一阶微分整体上构成一个算子值映射：
	\begin{equation*}
		\operatorname{D}T:E\to\mathcal{L}(X,Y),\quad x\to\operatorname{D}Tx
	\end{equation*}
	高阶微分所研究的正是一阶微分算子如何随基点$x$的变化而变化。既然$\operatorname{D}Tx$的取值属于算子空间$\mathcal{L}(X,Y)$，那么要研究它关于$x$的变化，就应当将$\operatorname{D}T$视为一个以$\mathcal{L}(X,Y)$为值域的映射，并对该映射再次求Fr\'echet微分，于是二阶微分自然满足：
	\begin{equation*}
		\operatorname{D}^2Tx\in\mathcal{L}\bigl(X,\mathcal{L}(X,Y)\bigr)
	\end{equation*}\par
	因此，微分层次的变化并不是人为规定的，而是由所研究对象的值域自然决定的：$T$的取值属于$Y$，所以$\operatorname{D}Tx$属于$\mathcal{L}(X,Y)$；而$\operatorname{D}T$的取值属于$\mathcal{L}(X,Y)$，所以$\operatorname{D}^2Tx$属于$\mathcal{L}\bigl(X,\mathcal{L}(X,Y)\bigr)$。更高阶微分可依此递归定义。
\end{note}
\begin{definition}
	设$X,Y$为实赋范线性空间，$E\subseteq X$是开集，$x_0\in E$，映射$T:E\to Y$。称：
	\begin{equation*}
		T(x_0+h)=Tx_0+Ah+r(h),\quad ||r(h)||_Y=\operatorname{o}(||h||_X),\;(||h||_X\to0)
	\end{equation*}
	为\textbf{无穷小增量公式}，其中$A\in\mathcal{L}(X,Y)$。
\end{definition}
\begin{definition}
	将所有在$E$上具有直到$n$次Fr\'echet微分并且每一阶微分都在$E$上连续的映射构成的集合记作$C^n(E)$。
\end{definition}
\begin{property}\label{prop:FrechetDifferential}
	设$X,Y$为实赋范线性空间，$E\subseteq X$是开集，$x_0\in E$，$T:E\to Y$和$S:E\to Y$是两个映射，则：
	\begin{enumerate}
		\item 若$T$在$x_0$处Fr\'echet可微，则$T$在$x_0$处的Fr\'echet微分是唯一的；
		\item 若$T$在$x_0$处Fr\'echet可微，则存在$x_0$的邻域$U$和$\alpha>0$使得：
		\begin{equation*}
			||Tx-Tx_0||_Y\leqslant\alpha||x-x_0||_X,\quad\forall\;x\in U
		\end{equation*}
		\item 若$T$在$x_0$处Fr\'echet可微，则$T$在$x_0$处连续；
		\item 若$T$在$x_0$处Fr\'echet可微，则$T$在$x_0$处具有无穷小增量公式；
		\item 若$T,S$在$x_0$处Fr\'echet可微，则对于任意的$\alpha,\beta\in\mathbb{R}$，$\alpha T+\beta S$在$x_0$处Fr\'echet可微，且有：
		\begin{equation*}
			\operatorname{D}(\alpha T+\beta S)x_0=\alpha \operatorname{D}Tx_0+\beta\operatorname{D}Sx_0
		\end{equation*}
		\item 设$Z$是实赋范线性空间，$F\subseteq Y$是开集，$R:F\to Z$是映射。若$T$在$x_0$处Fr\'echet可微，$TE\subseteq F$，$R$在$Tx_0$处Fr\'echet可微，则$RT:E\to Z$也在$x_0$处可微，且有：
		\begin{equation*}
			\operatorname{D}(RT)x_0=\operatorname{D}R(Tx_0)\circ\operatorname{D}Tx_0
		\end{equation*}
	\end{enumerate}
\end{property}
\begin{proof}
	(1)设$A,B\in\mathcal{L}(X,Y)$都是$T$在$x_0$处的Fr\'echet微分。任取$h\in X\setminus\{\mathbf{0}\}$，对于充分小的$t\ne0$有：
	\begin{equation*}
		||(B-A)h||_Y=\frac{1}{|t|}||(B-A)(th)||_Y
	\end{equation*}
	于是：
	\begin{align*}
		||(B-A)h||_Y\leqslant&\frac{||T(x_0+th)-Tx_0-A(th)||_Y}{|t|}+\frac{||T(x_0+th)-Tx_0-B(th)||_Y}{|t|} \\
		=&||h||_X\left[\frac{||T(x_0+th)-Tx_0-A(th)||_Y}{||th||_X}+\frac{||T(x_0+th)-Tx_0-B(th)||_Y}{||th||_X}\right]
	\end{align*}
	令$t\to0$，由Fr\'echet微分的定义可得$||(B-A)h||_Y=0$。根据$h$的任意性可知$A=B$。\par
	(2)设$\operatorname{D}Tx_0=A$。因为$T$在$x_0$处Fr\'echet可微，所以对于任意的$\varepsilon>0$，存在$\delta>0$使得当$0<||x-x_0||_X<\delta$时有：
	\begin{equation*}
		||Tx-Tx_0-A(x-x_0)||_Y<\varepsilon||x-x_0||_X
	\end{equation*}
	于是\info{算子范数}：
	\begin{align*}
		&||Tx-Tx_0||_Y\leqslant||Tx-Tx_0-A(x-x_0)||_Y+||A(x-x_0)||_Y \\
		<&\varepsilon||x-x_0||_X+||A||\;||x-x_0||_X=(\varepsilon+||A||)||x-x_0||_X
	\end{align*}
	取$\alpha=\varepsilon+||A||$即可。\par
	(3)由(2)立即可得。\par
	(4)由定义立即可得。\par
	(5)令$A=\operatorname{D}Tx_0,\;B=\operatorname{D}Sx_0$。对于任意的$h\in X\setminus\{\mathbf{0}\}$有：
	\begin{align*}
		&\frac{||[\alpha T(x_0+h)+\beta S(x_0+h)-\alpha Tx_0-\beta Sx_0]-(\alpha A+\beta B)h||_Y}{||h||_X} \\
		=&\frac{1}{||h||_X}\{||\alpha[T(x_0+h)-Tx_0-Ah]+\beta[S(x_0+h)-Sx_0-Bh]||_Y\} \\
		\leqslant&\frac{1}{||h||_X}\{||\alpha[T(x_0+h)-Tx_0-Ah]||_Y+||\beta[S(x_0+h)-Sx_0-Bh]||_Y\} \\
		=&|\alpha|\frac{||T(x_0+h)-Tx_0-Ah||_Y}{||h||_X}+|\beta|\frac{||S(x_0+h)-Sx_0-Bh||_Y}{||h||_X}
	\end{align*}
	于是由\cref{prop:RMap}(4)(5.b)(5.c)可得：
	\begin{align*}
		&0\leqslant\lim_{||h||_X\to0}\frac{||[\alpha T(x_0+h)+\beta S(x_0+h)-\alpha Tx_0-\beta Sx_0]-(\alpha A+\beta B)h||_Y}{||h||_X} \\
		\leqslant&|\alpha|\lim_{||h||_X\to0}\frac{||T(x_0+h)-Tx_0-Ah||_Y}{||h||_X}+|\beta|\lim_{||h||_X\to0}\frac{||S(x_0+h)-Sx_0-Bh||_Y}{||h||_X}=0
	\end{align*}
	所以：
	\begin{equation*}
		\lim_{||h||_X\to0}\frac{||[\alpha T(x_0+h)+\beta S(x_0+h)-\alpha Tx_0-\beta Sx_0]-(\alpha A+\beta B)h||_Y}{||h||_X}=0
	\end{equation*}
	根据(1)可知结论成立。\par
	(6)记：
	\begin{equation*}
		A=\operatorname{D}Tx_0\in\mathcal{L}(X,Y),\quad B=\operatorname{D}R(Tx_0)\in\mathcal{L}(Y,Z)
	\end{equation*}
	由于$T$在$x_0$处Fr\'echet可微，由(4)可得：
	\begin{equation*}
		T(x_0+h)-T(x_0)=Ah+r_1(h),\quad||r_1(h)||_Y=\operatorname{o}(||h||_X),\;(||h||_X\to0)
	\end{equation*}
	同理：
	\begin{equation*}
		R(Tx_0+k)-R(Tx_0)=Bk+r_2(k),\quad||r_2(k)||_Z=\operatorname{o}(||k||_Y),\;(||k||_Y\to0)
	\end{equation*}
	令$k=T(x_0+h)-T(x_0)$，于是：
	\begin{align*}
		&R(T(x_0+h))-R(Tx_0)=R(Tx_0+k)-R(Tx_0)\\
		=&Bk+r_2(k)=B[Ah+r_1(h)]+r_2(k)=BAh+Br_1(h)+r_2(k)
	\end{align*}
	因此：
	\begin{equation*}
		R(T(x_0+h))-R(Tx_0)-BAh=Br_1(h)+r_2(k)
	\end{equation*}
	当$k=0$时有$r_2(k)=0$，以下将含有$\dfrac{||r_2(k)||_Z}{||k||_Y}$的乘积约定为$0$。所以\info{算子范数}：
	\begin{align*}
		&\frac{||R(T(x_0+h))-R(Tx_0)-BAh||_Z}{||h||_X}=\frac{||Br_1(h)+r_2(k)||_Z}{||h||_X} \\
		\leqslant&\frac{||Br_1(h)||_Z}{||h||_X}+\frac{||r_2(k)||_Z}{||h||_X}\leqslant
		||B||\,\frac{||r_1(h)||_Y}{||h||_X}+\frac{||r_2(k)||_Z}{||k||_Y}\cdot\frac{||k||_Y}{||h||_X}
	\end{align*}
	由(2)可知存在常数$\alpha>0$使得：
	\begin{equation*}
		||T(x_0+h)-T(x_0)||_Y\leqslant\alpha||h||_X
	\end{equation*}
	即：
	\begin{equation*}
		\frac{||k||_Y}{||h||_X}\leqslant\alpha
	\end{equation*}
	由\cref{prop:RMap}(4)(5.b)(5.c)可得：
	\begin{equation*}
		\lim_{||h||_X\to0}\frac{||R(T(x_0+h))-R(Tx_0)-BAh||_Z}{||h||_X}=0\qedhere
	\end{equation*}
\end{proof}
\subsection{G\^ateaux微分}
\begin{definition}
	设$X,Y$为实赋范线性空间，$E\subseteq X$，$x_0\in E$，映射$T:E\to Y$。给定$h\in X$，若存在$\delta>0$使得$x_0+th\in E$对一切$t\in\mathbb{R},\;|t|<\delta$成立，
	并且存在极限：
	\begin{equation*}
		\lim_{t\to0}\frac{T(x_0+th)-Tx_0}{t}
	\end{equation*}
	则称该极限为映射$T$在点$x_0$沿方向$h$的G\^ateaux \gls{DirectionalDerivative}，记为$\dif T(x_0;h)$。若对任意的$h\in X$，$\dif T(x_0;h)$都存在且存在有界线性算子$A\in\mathcal{L}(X,Y)$满足：
	\begin{equation*}
		\forall\;h\in X,\;\dif T(x_0;h)=Ah
	\end{equation*}
	则称$T$在$x_0$处G\^ateaux可微，称$A$是$T$在点$x_0$处的G\^ateaux微分，记作$\operatorname{D}_GTx_0$。
\end{definition}
\begin{definition}
	设$X,Y$为实赋范线性空间，$E\subseteq X$，$x_0\in E$，映射$T:E\to Y$。若对给定的$h\in X$，映射$x\to\dif T(x;h)$在点$x_0$处沿方向$h_1\in X$的G\^ateaux方向导数存在，即极限：
	\begin{equation*}
		\lim_{t\to0}\frac{\dif T(x_0+th_1;h)-\dif T(x_0;h)}{t}
	\end{equation*}
	则称该极限为映射$T$在点$x_0$沿方向$h,h_1$的二阶G\^ateaux方向导数，记作$\dif^2T(x_0;h,h_1)$。若对任意$h,h_1\in X$，二阶G\^ateaux方向导数$\dif^2T(x_0;h,h_1)$均存在，并且存在有界线性算子$B\in\mathcal{L}\bigl(X,\mathcal{L}(X,Y)\bigr)$使得：
	\begin{equation*}
		\forall\;h,h_1\in X,\;\dif^2T(x_0;h,h_1)=(Bh_1)h
	\end{equation*}
	则称$T$在点$x_0$处二阶G\^ateaux可微，称$B$为$T$在点$x_0$处的
	二阶G\^ateaux微分，记作$\operatorname{D}_G^2Tx_0$。更高阶的G\^ateaux方向导数与G\^ateaux微分可依此类推。
\end{definition}
\begin{note}
	一阶G\^ateaux微分$\operatorname{D}_GTx$描述映射$T$在点$x$附近沿所有方向发生微小变化时，映射$T$所产生的一阶变化。二阶G\^ateaux微分进一步研究的不是$T$本身如何变化，而是研究一阶微分算子$\operatorname{D}_GTx$如何随基点$x$的变化而变化。
\end{note}
\begin{property}\label{prop:FrechetGateaux}
	设$X,Y$为实赋范线性空间，$E\subseteq X$是开集，$x_0\in E$，$T:E\to Y$是映射。Fr\'echet可微和G\^ateaux可微有如下关系：
	\begin{enumerate}
		\item 若$T$在$x_0$处Fr\'echet可微，则$T$在$x_0$处G\^ateaux可微，且$\operatorname{D}_GTx_0=\operatorname{D}Tx_0$；
		\item 若$X=Y=\mathbb{R}$，则Fr\'echet可微和G\^ateaux可微等价，且$\operatorname{D}_GTx_0=\operatorname{D}Tx_0$；
	\end{enumerate}
\end{property}
\begin{proof}
	(1)任取$h\in X\setminus\{\mathbf{0}\}$，由\cref{prop:FrechetDifferential}(4)可得：
	\begin{equation*}
		\frac{T(x_0+th)-Tx_0}{t}=\frac{(\operatorname{D}Tx_0)(th)+r(th)}{t}=(\operatorname{D}Tx_0)h+\frac{r(th)}{t}
	\end{equation*}
	其中$||r(h)||_Y=\operatorname{o}(||h||_X)(||h||_X\to0)$。注意到：
	\begin{equation*}
		\left\|\frac{r(th)}{t}\right\|_Y=\frac{||r(th)||_Y}{|t|}=\frac{||r(th)||_Y}{||th||_X}||h||_X
	\end{equation*}
	由\cref{prop:RMap}(5.c)可得：
	\begin{equation*}
		\lim_{t\to0}\left\|\frac{r(th)}{t}\right\|_Y=\lim_{t\to0}\frac{||r(th)||_Y}{||th||_X}||h||_X=||h||_X\lim_{t\to0}\frac{||r(th)||_Y}{||th||_X}=0
	\end{equation*}
	所以根据\cref{prop:Norm}(3)可得：
	\begin{equation*}
		\lim_{t\to0}\frac{T(x_0+th)-Tx_0}{t}=(\operatorname{D}Tx_0)h+\lim_{t\to0}\frac{r(th)}{t}=(\operatorname{D}Tx_0)h
	\end{equation*}
	当$h=\mathbf{0}$时结论显然成立。\par
	(2)由(1)可知只需证明此时若$T$在$x_0$处G\^ateaux可微，则$T$在$x_0$处Fr\'echet可微。\par
	因为$T$在$x_0$处G\^ateaux可微，所以对于任意的$h\in\mathbb{R}$，$\dif T(x_0;h)$都存在且存在有界线性算子$A\in\mathcal{L}(\mathbb{R},\mathbb{R})$满足：
	\begin{equation*}
		\forall\;h\in\mathbb{R},\;\dif T(x_0;h)=Ah
	\end{equation*}
	因为$A$是从$\mathbb{R}$到$\mathbb{R}$上的有界线性算子，所以$Ah=a\times h$，其中$a\in\mathbb{R}$。在G\^ateaux可微的定义式中取$h=1$即可得到：
	\begin{equation*}
		\lim_{t\to0}\frac{T(x_0+t)-Tx_0}{t}=a
	\end{equation*}
	由\cref{prop:RSeq}(8.a)(8.b)可得：
	\begin{align*}
		\lim_{h\to0}\frac{|T(x_0+h)-Tx_0-Ah|}{|h|}&=\lim_{h\to0}\frac{|T(x_0+h)-Tx_0-ah|}{|h|}=\lim_{h\to0}\left|\frac{T(x_0+h)-Tx_0}{h}-a\right| \\
		&=\left|\lim_{h\to0}\left[\frac{T(x_0+h)-Tx_0}{h}-a\right]\right|=0\qedhere
	\end{align*}
\end{proof}
\begin{note}
	根据\cref{prop:FrechetGateaux}(2)，当$X=Y=\mathbb{R}$时，简称$T$ Fr\'echet可微或$T$ G\^ateaux可微为$T$可微，对应的Fr\'echet微分或G\^ateaux微分记作$T'$，二阶微分记作$T''$或$T^{(2)}$，更高阶的情况可依此类推。
\end{note}
\subsection{偏微分与偏导数}
\begin{definition}
	设$\seq{X}{n}$均为实赋范线性空间。在乘积空间：
	\begin{equation*}
		X=X_1\times X_2\times\cdots\times X_n
	\end{equation*}
	上定义范数：
	\begin{equation*}
		||(x^1,x^2,\dots,x^n)||=\sum_{i=1}^{n}||x^i||_{X_i}
	\end{equation*}
	$Y$为实赋范线性空间，$E\subseteq X$为开集，$x_0=(x_0^1,x_0^2,\dots,x_0^n)\in E$，映射$T:E\to Y$。对于$i=1,2,\dots,n$，记
	\begin{gather*}
		E_{x_0,i}=\{u^i\in X_i:(x_0^1,\dots,x_0^{i-1},u^i,x_0^{i+1},\dots,x_0^n)\in E\} \\
		T_{x_0,i}(u^i)=T(x_0^1,\dots,x_0^{i-1},u^i,x_0^{i+1},\dots,x_0^n)
	\end{gather*}
	若$T_{x_0,i}$在$x_0^i$处Fr\'echet可微，则称其Fr\'echet微分为$T$在$x_0$处关于第$i$个自变量的偏Fr\'echet微分，记作$\operatorname{D}_iTx_0$。偏G\^ateaux微分$\operatorname{D}_{G,i}Tx_0$以及偏G\^ateaux方向导数$\dif_iT(x_0;h_i)$可类似定义，其中$h_i\in X_i$。
\end{definition}
\begin{property}\label{prop:PartialDifferential}
	在上述定义中，记：
	\begin{equation*}
		\iota_i:X_i\to X,\quad\iota_ih_i=(\mathbf{0},\dots,\mathbf{0},h_i,\mathbf{0},\dots,\mathbf{0})
	\end{equation*}
	则有以下结论：
	\begin{enumerate}
		\item 若$T$在$x_0$处Fr'echet可微，则$T$在$x_0$处关于每一个自变量都偏Fr'echet可微，并且：
		\begin{equation*}
			\operatorname{D}_iTx_0=(\operatorname{D}Tx_0)\circ\iota_i
		\end{equation*}
		对于任意的$h=(h_1,h_2,\dots,h_n)\in X$还有：
		\begin{equation*}
			\operatorname{D}Tx_0(h)=\sum_{i=1}^{n}\operatorname{D}_iTx_0(h_i)
		\end{equation*}
		\item 若$T$在$x_0$处G\^ateaux可微，则$T$在$x_0$处关于每一个自变量都偏G\^ateaux可微，并且：
		\begin{equation*}
			\operatorname{D}_{\mathrm{G},i}Tx_0=(\operatorname{D}_{\mathrm{G}}Tx_0)\circ\iota_i
		\end{equation*}
		对于任意的$h=(h_1,h_2,\dots,h_n)\in X$还有：
		\begin{equation*}
			\operatorname{D}_{\mathrm{G}}Tx_0(h)=\sum_{i=1}^{n}\operatorname{D}_{\mathrm{G},i}Tx_0(h_i)
		\end{equation*}
	\end{enumerate}
\end{property}
\begin{proof}
	(1)由\cref{prop:FrechetDifferential}(4)可得：
	\begin{equation*}
		T(x_0+h)-Tx_0=(\operatorname{D}Tx_0)h+r(h),\quad||r(h)||_Y=\operatorname{o}(||h||_X)
	\end{equation*}
	固定$i$，注意到$||\iota_ih_i||_X=||h_i||_{X_i}$，所以：
	\begin{equation*}
		T_{x_0,i}(x_0^i+h_i)-T_{x_0,i}(x_0^i)-(\operatorname{D}Tx_0)\iota_ih_i=r(\iota_ih_i)=\operatorname{o}(||h_i||_{X_i})
	\end{equation*}
	由偏Fr\'echet微分的定义和\cref{prop:FrechetDifferential}(1)可得：
	\begin{equation*}
		\operatorname{D}_iTx_0=\operatorname{D}Tx_0\circ\iota_i
	\end{equation*}
	对于任意的$h=(h_1,h_2,\dots,h_n)\in X$有$h=\sum\limits_{i=1}^{n}\iota_ih_i$，结合$\operatorname{D}Tx_0$的线性性可得：
	\begin{equation*}
		\operatorname{D}Tx_0(h)=\sum_{i=1}^{n}\operatorname{D}Tx_0(\iota_ih_i)=\sum_{i=1}^{n}\operatorname{D}_iTx_0(h_i)
	\end{equation*}\par
	(2)固定$i\in\{1,2,\dots,n\}$。由于$T$在$x_0$处G\^ateaux可微，所以对于任意的$h_i\in X_i$有：
	\begin{align*}
		\lim_{t\to0}\frac{T_{x_0,i}(x_0^i+th_i)-T_{x_0,i}(x_0^i)}{t}=\lim_{t\to0}\frac{T(x_0+t\iota_ih_i)-Tx_0}{t}=\operatorname{D}_{G}Tx_0(\iota_ih_i)
	\end{align*}
	可以验证得到$(\operatorname{D}_{G}Tx_0)\circ\iota_i:X_i\to Y$是有界线性算子，于是由偏G\^ateaux微分的定义可得$T$在$x_0$处关于第$i$个自变量偏G\^ateaux可微，并且：
	\begin{equation*}
		\operatorname{D}_{G,i}Tx_0=(\operatorname{D}_{G}Tx_0)\circ\iota_i
	\end{equation*}
	对于任意的$h=(h_1,h_2,\dots,h_n)\in X$有$h=\sum\limits_{i=1}^{n}\iota_ih_i$，结合$\operatorname{D}_{G}Tx_0$的线性性可得：
	\begin{equation*}
		\operatorname{D}_{G}Tx_0(h)=\operatorname{D}_{G}Tx_0\left(\sum_{i=1}^{n}\iota_ih_i\right)=\sum_{i=1}^{n}\operatorname{D}_{\mathrm{G}}Tx_0(\iota_ih_i)=\sum_{i=1}^{n}\operatorname{D}_{\mathrm{G},i}Tx_0(h_i)\qedhere
	\end{equation*}
\end{proof}
\begin{definition}
	在之前的定义中，设$X_i$是一维实赋范线性空间，并指定非零向量$v_i\in X_i$。若$T$在$x_0$处关于第$i$个自变量偏G\^ateaux可微，则称：
	\begin{equation*}
		\partial_{i,v_i}Tx_0\coloneq\operatorname{D}_{G,i}Tx_0(v_i)=\dif_iT(x_0;v_i)
	\end{equation*}
	为$T$在$x_0$处关于第$i$个自变量相对于基向量$v_i$的\gls{PartialDerivative}。
\end{definition}
\begin{note}
	本节按照对象所属的空间区分称呼：$\operatorname{D}_{G,i}Tx_0\in\mathcal{L}(X_i,Y)$是有界线性算子，称为偏G\^ateaux微分；$\partial_{i,v_i}Tx_0=\operatorname{D}_{G,i}Tx_0(v_i)\in Y$是该算子在指定基向量上的取值，称为偏导数。偏G\^ateaux微分本身已经完整刻画了映射$T$在$x_0$处关于第$i$个自变量的一阶变化；在$X_i$为一维空间时进一步引入相对于$v_i$的偏导数，是为了将算子值的偏微分表示为$Y$中的单个元素。\par
	事实上，由$X_i=\operatorname{span}\{v_i\}$可知，对于任意$h_i\in X_i$，存在唯一的$\lambda\in\mathbb{R}$使得$h_i=\lambda v_i$。由$\operatorname{D}_{G,i}Tx_0$的线性性可得：
	\begin{align*}
		\operatorname{D}_{G,i}Tx_0(h_i)=\operatorname{D}_{G,i}Tx_0(\lambda v_i)=\lambda\operatorname{D}_{G,i}Tx_0(v_i)
	\end{align*}
	所以，选定$v_i$之后，整个偏G\^ateaux微分都由偏导数$\partial_{i,v_i}Tx_0\in Y$唯一确定。\par
	抽象的一维实赋范线性空间通常没有天然指定的基向量，而$\mathbb{R}$作为自身上的一维实赋范线性空间具有标准基向量$1$。因此，当$X_i=\mathbb{R}$时不需要另行选择基向量，并将相应的偏导数简记为：
	\begin{equation*}
		\partial_iTx_0\coloneq\partial_{i,1}Tx_0
		=\operatorname{D}_{G,i}Tx_0(1)=\dif_iT(x_0;1)
	\end{equation*}
	若将第$i$个自变量记作$x^i$，也使用常见记法：
	\begin{equation*}
		\frac{\partial T}{\partial x^i}(x_0)\coloneq\partial_iTx_0,\quad
		\frac{\partial^2T}{\partial x^j\partial x^i}(x_0)\coloneq\partial_j\partial_iTx_0
	\end{equation*}
	其中第二个等式要求相应的二阶偏导数存在。
\end{note}

\subsection{实值函数的微分}
\subsubsection{有限增量公式}
\begin{definition}
	设$X$为实赋范线性空间，开集$E\subseteq X$，函数$f:E\to\mathbb{R}$，$x_0\in E$。若存在$x_0$的邻域$U(x_0,\delta)\subseteq E$使得对任意的$x\in U(x_0,\delta)$有：
	\begin{equation*}
		f(x)\leqslant f(x_0)\quad\text{或}\quad f(x)\geqslant f(x_0)
	\end{equation*}
	则称$f(x)$在$x_0$处取到\textbf{极大值}或\textbf{极小值}。若上述不等号对任意的$x\in U(x_0,\delta)\setminus\{x_0\}$严格地成立，则称$f(x)$在$x_0$处取到\textbf{严格的极大值}或\textbf{严格的极小值}。函数的极大值与极小值统称为\textbf{极值}，使函数取到极值的点$x\in E$被称为函数的\textbf{极值点}。
\end{definition}
\begin{lemma}\label{lem:SmallOSignPreservation}
	设函数$f:\mathbb{R}\to\mathbb{R}$，$A\in\mathbb{R}\setminus\{0\}$。若$f(h)=Ah+\operatorname{o}(h)(h\to0)$，那么对于充分小的$h\ne0$，$f(h)$与$Ah$同号。
\end{lemma}
\begin{proof}
	由\cref{prop:RMap}(5.b)(5.c)可得：
	\begin{equation*}
		\lim_{h\to0}\frac{f(h)}{Ah}=\lim_{h\to0}\frac{Ah}{Ah}+\lim_{h\to0}\frac{\operatorname{o}(h)}{Ah}=1+\frac{1}{A}\lim_{h\to0}\frac{\operatorname{o}(h)}{h}
	\end{equation*}
	根据\cref{prop:RMap}(3)即可得到结论。
\end{proof}
\begin{theorem}[Fermat Theorem]\label{theo:FermatTheorem}
	设函数$f:I\to\mathbb{R}$，开区间$I\subseteq\mathbb{R}$，$f$在$x_0\in I$处取得极值。若$f$在$x_0$处可微，则有$f'(x_0)=0$。
\end{theorem}
\begin{proof}
	由\cref{prop:FrechetDifferential}(4)可得：
	\begin{equation*}
		f(x)-f(x_0)=f'(x_0)(x-x_0)+\operatorname{o}(|x-x_0|),\quad(|x-x_0|\to0)
	\end{equation*}
	根据\cref{lem:SmallOSignPreservation}即可得出结论。
\end{proof}
\begin{theorem}[Rolle Theorem]\label{theo:RolleTheorem}
	设$\mathbb{R}$上的函数$f$在$[a,b]$上连续，$(a,b)$上可微，$f(a)=f(b)$，则存在$c\in(a,b)$使得$f'(c)=0$。
\end{theorem}
\begin{proof}
	由\cref{prop:OpenClosedSet}(2)、\cref{theo:CompactRn}和\cref{prop:CompactMap}(3)可知$f$在$[a,b]$上取到最小值$m\in\mathbb{R}$和最大值$M\in\mathbb{R}$。若$M=m$，则$f$在$[a,b]$上是常值函数，由\cref{theo:FermatTheorem}可知对任意的$c\in(a,b)$都有$f'(c)=0$。若$M\ne m$，则存在$c\in(a,b)$使得$f$在$c$处取得极值。由\cref{theo:FermatTheorem}即可得出结论。
\end{proof}
\begin{theorem}[Lagrange Mean Value Theorem]\label{theo:LagrangeMeanValueTheorem}
	设$\mathbb{R}$上的函数$f$在$[a,b]$上连续，$(a,b)$上可微，则存在$c\in(a,b)$使得：
	\begin{equation*}
		f'(c)=\frac{f(b)-f(a)}{b-a}
	\end{equation*}
	即存在$\theta\in(0,1)$使得：
	\begin{equation*}
		f(b)-f(a)=f'[a+\theta(b-a)](b-a)
	\end{equation*}
\end{theorem}
\begin{proof}
	作辅助函数：
	\begin{equation*}
		F(x)=f(x)-f(a)-\frac{f(b)-f(a)}{b-a}(x-a)
	\end{equation*}
	由\cref{prop:RMap}(5.b)(5.c)和\cref{prop:FrechetDifferential}(5)可知$F(x)$在$[a,b]$上连续，$(a,b)$上可微，$F(a)=F(b)=0$。根据\cref{theo:RolleTheorem}可知存在$c\in(a,b)$使得：
	\begin{equation*}
		0=F'(c)=f'(c)-\frac{f(b)-f(a)}{b-a}\qedhere
	\end{equation*}
\end{proof}
\begin{theorem}[有限增量公式]\label{theo:FiniteIncrement}
	设$X$为实赋范线性空间，$E\subseteq X$为开集，$f:E\to\mathbb{R}$在连接$a,b\in E$的线段$\{a+t(b-a):t\in[0,1]\}$上连续，并在开线段$\{a+t(b-a):t\in(0,1)\}$上Fr\'echet可微，则存在$\theta\in(0,1)$使得：
	\begin{equation*}
		f(b)-f(a)=\operatorname{D}f[a+\theta(b-a)](b-a)
	\end{equation*}
\end{theorem}
\begin{proof}
	考察函数：
	\begin{equation*}
		\varphi(t)=f[a+t(b-a)]
	\end{equation*}
	$\varphi(t)$在$[0,1]$上连续，在$(0,1)$上可微，由\cref{theo:LagrangeMeanValueTheorem}可得：
	\begin{equation*}
		\varphi(1)-\varphi(0)=\varphi'(\theta),\;\theta\in(0,1)
	\end{equation*}
	根据\cref{prop:FrechetDifferential}(6)可知：
	\begin{equation*}
		\varphi'(t)=\operatorname{D}f[a+t(b-a)](b-a)
	\end{equation*}
	于是：
	\begin{equation*}
		\varphi(1)-\varphi(0)=f(b)-f(a)=\operatorname{D}f[a+\theta(b-a)](b-a)\qedhere
	\end{equation*}
\end{proof}
\subsubsection{Taylor公式}
\begin{theorem}[Cauchy Mean Value Theorem]\label{theo:CauchyMeanValueTheorem}
	设$\mathbb{R}$上的函数$f,g$在$[a,b]$上连续，$(a,b)$上可微，且对任意的$x\in(a,b)$有$g'(x)\ne0$，则存在$c\in(a,b)$使得：
	\begin{equation*}
		\frac{f(b)-f(a)}{g(b)-g(a)}=\frac{f'(c)}{g'(c)}
	\end{equation*}
	即存在$\theta\in(0,1)$使得：
	\begin{equation*}
		\frac{f(b)-f(a)}{g(b)-g(a)}=\frac{f'[a+\theta(b-a)]}{g'[a+\theta(b-a)]}
	\end{equation*}
\end{theorem}
\begin{proof}
	由\cref{theo:LagrangeMeanValueTheorem}可知$g(b)-g(a)\ne0$，定义函数：
	\begin{equation*}
		F(x)=f(x)-f(a)-\frac{f(b)-f(a)}{g(b)-g(a)}[g(x)-g(a)]
	\end{equation*}
	由\cref{prop:RMap}(5.b)(5.c)和\cref{prop:FrechetDifferential}(5)可知$F(x)$在$[a,b]$上连续，$(a,b)$上可微，$F(a)=F(b)=0$。根据\cref{theo:RolleTheorem}可知存在$c\in(a,b)$使得：
	\begin{equation*}
		0=F'(c)=f'(c)-\frac{f(b)-f(a)}{g(b)-g(a)}g'(c)\qedhere
	\end{equation*}
\end{proof}
\begin{lemma}\label{lem:Taylor1}
	设$\mathbb{R}$上的函数$f$在$U(a,\delta)$上有定义，$f$在点$a$处$n$阶可微。若$f(a)=f'(a)=f''(a)=\cdots=f^{(n)}(a)=0$，则$f(x)=\operatorname{o}\Big((x-a)^n\Big)(x\to a)$。
\end{lemma}
\begin{proof}
	由\info{洛必达}和\cref{prop:RMap}(5.c)可得：
	\begin{align*}
		&\lim_{x\to a}\frac{f(x)}{(x-a)^n}=\lim_{x\to a}\frac{f'(x)}{n(x-a)^{n-1}}=\lim_{x\to a}\frac{f''(x)}{n(n-1)(x-a)^{n-2}}=\cdots=\lim_{x\to a}\frac{f^{(n-1)}(x)}{(n-1)!(x-a)} \\
		=&\lim_{x\to a}\frac{f^{(n-1)}(x)-f^{(n-1)}(a)}{(n-1)!(x-a)}=\frac{1}{(n-1)!}\lim_{x\to a}\frac{f^{(n-1)}(x)-f^{(n-1)}(a)}{(x-a)}=\frac{f^{(n)}(a)}{(n-1)!}=0\qedhere
	\end{align*}
\end{proof}
\begin{lemma}\label{lem:Taylor2}
	设$\mathbb{R}$上的函数$f,g$在$U(a,\delta)$上有定义，$f,g$在点$a$处$n$阶可微。若$f^{(i)}(a)=g^{(i)}(a),\;i=0,1,\dots,n$，则$f(x)=g(x)+\operatorname{o}\Big((x-a)^n\Big)(x\to a)$。
\end{lemma}
\begin{proof}
	在\cref{lem:Taylor1}中取$f-g$即可得到。
\end{proof}
\begin{lemma}\label{lem:Taylor3}
	设$\mathbb{R}$上的函数$f$在$U(a,\delta)$上有定义，$f$在点$a$处$n$阶可微，则：
	\begin{equation*}
		f(x)=\sum_{i=0}^{n}\frac{f^{(i)}(a)}{i!}(x-a)^i+\operatorname{o}\Big((x-a)^n\Big),\;(x\to a)
	\end{equation*}
\end{lemma}
\begin{proof}
	令：
	\begin{equation*}
		g(x)=\sum_{i=0}^{n}\frac{f^{(i)}(a)}{i!}(x-a)^i
	\end{equation*}
	则对任意$i=0,1,\dots,n$，都有$g^{(i)}(a)=f^{(i)}(a)$。由\cref{lem:Taylor2}可得：
	\begin{equation*}
		f(x)=g(x)+\operatorname{o}\Big((x-a)^n\Big),\;(x\to a)\qedhere
	\end{equation*}
\end{proof}
\begin{lemma}\label{lem:Taylor4}
	设$\mathbb{R}$上的函数$f\in C^n([a,b])$，$f$在$(a,b)$上$n+1$阶可微，$y\in [a,b]$。若$f(y)=f'(y)=f''(y)=\cdots=f^{(n)}(y)=0$，则对任意的$x\in[a,b]$，存在介于$x$和$y$之间的$c$使得：
	\begin{equation*}
		f(x)=\frac{f^{(n+1)}(c)}{(n+1)!}(x-y)^{n+1}
	\end{equation*}
\end{lemma}
\begin{proof}
	记$g(x)=(x-y)^{n+1}$，则$g(y)=g'(y)=g''(y)=\cdots=g^{(n)}(y)=0$。由\cref{theo:ElementaryContinuity}和\cref{theo:CauchyMeanValueTheorem}可知存在介于$x$和$y$之间的$c_1$使得：
	\begin{equation*}
		\frac{f(x)}{g(x)}=\frac{f(x)-f(y)}{g(x)-g(y)}=\frac{f'(c_1)}{g'(c_1)}
	\end{equation*}
	由\cref{theo:ElementaryContinuity}和\cref{theo:CauchyMeanValueTheorem}可知存在介于$c_1$和$y$之间的$c_2$使得：
	\begin{equation*}
		\frac{f'(c_1)}{g'(c_1)}=\frac{f'(c_1)-f'(y)}{g'(c_1)-g'(y)}=\frac{f''(c_2)}{g''(c_2)}
	\end{equation*}
	依此类推可得：
	\begin{equation*}
		\frac{f(x)}{g(x)}=\frac{f'(c_1)}{g'(c_1)}=\frac{f''(c_2)}{g''(c_2)}=\cdots=\frac{f^{(n)}(c_n)}{g^{(n)}(c_n)}=\frac{f^{(n+1)}(c)}{g^{(n+1)}(c)}
	\end{equation*}
	即：
	\begin{equation*}
		f(x)=\frac{f^{(n+1)}(c)}{(n+1)!}(x-y)^{n+1}\qedhere
	\end{equation*}
\end{proof}
\begin{lemma}\label{lem:Taylor5}
	设$\mathbb{R}$上的函数$f\in C^n([a,b])$，$f$在$(a,b)$上$n+1$阶可微，$y\in [a,b]$，则对任意的$x\in[a,b]$，存在介于$x$和$y$之间的$c$使得：
	\begin{equation*}
		f(x)=\sum_{i=0}^{n}\frac{f^{(i)}(y)}{i!}(x-y)^i+\frac{f^{(n+1)}(c)}{(n+1)!}(x-y)^{n+1}
	\end{equation*}
\end{lemma}
\begin{proof}
	对函数：
	\begin{equation*}
		f(x)-\sum_{i=0}^{n}\frac{f^{(i)}(y)}{i!}(x-y)^i
	\end{equation*}
	使用\cref{theo:ElementaryContinuity}、\cref{prop:RMap}(5.b)(5.c)、\cref{prop:FrechetDifferential}(5)和\cref{lem:Taylor4}即可得到。
\end{proof}
\begin{theorem}[Taylor's Theorem]\label{theo:TaylorTheorem}
	称以下公式为Taylor公式：
	\begin{enumerate}
		\item(Peano余项) 设$X$为实赋范线性空间，$E\subseteq X$为开集，函数$f:E\to\mathbb{R}$在$x_0\in E$处$n$阶Fr\'echet可微。对任意$h\in X$有：
		\begin{equation*}
			f(x_0+th)=\sum_{i=0}^{n}\frac{t^i}{i!}\operatorname{D}^ifx_0(\underbrace{h,\dots,h}_{i\text{个}h})+\operatorname{o}(t^n),\;(t\to0)
		\end{equation*}
		\item(Lagrange余项) 设$X$为实赋范线性空间，$E\subseteq X$为开集，$x_0\in E$，函数$f:E\to\mathbb{R}$在$\{x_0+th:0\leqslant t\leqslant1\}$上$n+1$次Fr\'echet可微，并且$\{x_0+th:0\leqslant t\leqslant1\}\subseteq E$，则存在$\theta\in(0,1)$使得
		\begin{equation*}
			f(x_0+h)=\sum_{i=0}^{n}\frac{1}{i!}\operatorname{D}^ifx_0(\underbrace{h,\dots,h}_{i\text{个}h})
			+\frac{1}{(n+1)!}\operatorname{D}^{n+1}f(x_0+\theta h)(\underbrace{h,\dots,h}_{n+1\text{个}h})
		\end{equation*}
	\end{enumerate}
\end{theorem}
\begin{proof}
	(1)考察函数：
	\begin{equation*}
		\varphi(t)=f(x_0+th)
	\end{equation*}
	由于$f$在$x_0$处$n$次Fr\'echet可微，所以$\varphi$在$t=0$处$n$次可微。由\cref{prop:FrechetDifferential}(6)可得对$i=1,\dots,n$有：
	\begin{equation*}
		\varphi^{(i)}(0)=\operatorname{D}^ifx_0(\underbrace{h,\dots,h}_{i\text{个}h})
	\end{equation*}
	根据\cref{lem:Taylor3}可得：
	\begin{equation*}
		\varphi(t)=\sum_{i=0}^{n}\frac{\varphi^{(i)}(0)}{i!}t^i+\operatorname{o}(t^n)=\sum_{i=0}^{n}\frac{t^i}{i!}\operatorname{D}^ifx_0(\underbrace{h,\dots,h}_{i\text{个}h})+\operatorname{o}(t^n),\;(t\to0)
	\end{equation*}\par
	(2)考察函数：
	\begin{equation*}
		\varphi(t)=f(x_0+th)
	\end{equation*}
	由\cref{prop:FrechetDifferential}(3)可知$\varphi\in C^n([0,1])$，且在$[0,1]$上$n+1$阶可微。由\cref{prop:FrechetDifferential}(6)可得对$i=1,\dots,n$有：
	\begin{equation*}
		\varphi^{(i)}(0)=\operatorname{D}^ifx_0(\underbrace{h,\dots,h}_{i\text{个}h})
	\end{equation*}
	根据\cref{lem:Taylor5}可知存在$\theta\in(0,1)$使得：
	\begin{align*}
		&\varphi(1)=\sum_{i=0}^{n}\frac{\varphi^{(i)}(0)}{i!}
		+\frac{\varphi^{(n+1)}(\theta)}{(n+1)!} \\
		=&\sum_{i=0}^{n}\frac{1}{i!}\operatorname{D}^ifx_0(\underbrace{h,\dots,h}_{i\text{个}h})
		+\frac{1}{(n+1)!}\operatorname{D}^{n+1}f(x_0+\theta h)(\underbrace{h,\dots,h}_{n+1\text{个}h})\qedhere
	\end{align*}
\end{proof}
\subsubsection{极值}
\begin{theorem}\label{theo:FrechetExtremumOptimality}
	设$X$为实赋范线性空间，$E\subseteq X$为开集，函数$f:E\to\mathbb{R}$在$x_0\in E$处Fr\'echet可微。
	\begin{enumerate}
		\item(一阶必要条件) 若$x_0$是$f$的极值点，则$\operatorname{D}fx_0=0$；
		\item(二阶必要条件) 若$f$在$x_0$处二阶Fr\'echet可微，$x_0$是$f$的极小（大）值点，则$\operatorname{D}fx_0=0$且对任意$h\in X$都有$\operatorname{D}^2fx_0(h,h)\geqslant0$（$\operatorname{D}^2fx_0(h,h)\leqslant0$）；
		\item(二阶充分条件) 若$f$在$x_0$处二阶Fr\'echet可微，$\operatorname{D}fx_0=0$，且存在常数$c>0$，使得对任意$h\in X$都有$\operatorname{D}^2fx_0(h,h)\geqslant c||h||_X^2$（$\operatorname{D}^2fx_0(h,h)\leqslant-c||h||_X^2$），则$x_0$是$f$的严格极小（大）值点。
	\end{enumerate}
\end{theorem}
\begin{proof}
	只证极小值情形，极大值完全类似。\par
	(1)任取$h\in X$，考察函数：
	\begin{equation*}
		\varphi(t)=f(x_0+th)
	\end{equation*}
	由于$E$是开集，故当$t$充分接近$0$时，$x_0+th\in E$，从而$\varphi$在$0$附近有定义。\par
	因为$x_0$是$f$的极小值点，所以存在$\delta>0$，使得当$||x-x_0||<\delta$时有$f(x)\geqslant f(x_0)$，于是对充分小的$|t|$，若$||th||<\delta$，则：
	\begin{equation*}
		\varphi(t)=f(x_0+th)\geqslant f(x_0)=\varphi(0)
	\end{equation*}
	因此$t=0$是$\varphi$的极小值点。\par
	对任意的$h\in X$，由\cref{prop:FrechetDifferential}(4)可得：
	\begin{equation*}
		f(x_0+th)=f(x_0)+\operatorname{D}fx_0(th)+r(th),\quad|r(th)|=\operatorname{o}(||th||_X),\;(||th||_X\to0)
	\end{equation*}
	由\cref{prop:RMap}(5.c)可得：
	\begin{align*}
		&\lim_{|t|\to0}\frac{|\varphi(t)-\varphi(0)-\operatorname{D}fx_0(th)|}{|t|}=\lim_{|t|\to0}\frac{|f(x_0+th)-f(x_0)-\operatorname{D}fx_0(th)|}{|t|} \\
		=&\lim_{|t|\to0}\frac{|\operatorname{D}fx_0(th)+r(th)-\operatorname{D}fx_0(th)|}{|t|}=\lim_{|t|\to0}\frac{|r(th)|}{|t|}=\lim_{|t|\to0}\frac{|r(th)|}{||th||_X}\frac{||th||_X}{|t|} \\
		=&||h||_X\lim_{|t|\to0}\frac{|r(th)|}{||th||_X}=||h||_X\lim_{||th||_X\to0}\frac{|r(th)|}{||th||_X}=0
	\end{align*}
	即$\varphi$在$t=0$处可微，且$\varphi'(0)=(\operatorname{D}fx_0)h$。\par
	由\cref{theo:FermatTheorem}可得$\varphi'(0)=0$，根据$h$的任意性可知$\operatorname{D}fx_0=0$。\par
	(2)由(1)即可得到$\operatorname{D}fx_0=0$。\par
	任取$h\in X$，考察函数：
	\begin{equation*}
		\varphi(t)=f(x_0+th)
	\end{equation*}
	由于$x_0$是极小值点，故$t=0$是$\varphi$的极小值点。因为$f$在$x_0$处二阶Fr\'echet可微，所以由\cref{theo:TaylorTheorem}(1)可得
	\begin{equation*}
		f(x_0+th)=f(x_0)+t(\operatorname{D}fx_0)h+\frac{t^2}{2}\operatorname{D}^2fx_0(h,h)+\operatorname{o}(t^2),\;(t\to0)
	\end{equation*}
	因为$t=0$是极小值点，所以对充分小的$t\ne0$有：
	\begin{equation*}
		0\leqslant\frac{\varphi(t)-\varphi(0)}{t^2}
		=\frac{1}{2}\operatorname{D}^2fx_0(h,h)+\frac{\operatorname{o}(t^2)}{t^2}
	\end{equation*}
	令$t\to0$即可得到$\operatorname{D}^2fx_0(h,h)\geqslant0$。\par
	(3)因为$f$在$x_0$处二阶Fr\'echet可微，所以：
	\begin{equation*}
		\operatorname{D}f(x_0+h)=\operatorname{D}fx_0+\operatorname{D}^2fx_0(h,\cdot)+R(h),\quad
		||R(h)||=\operatorname{o}(||h||_X)
	\end{equation*}
	于是存在$\delta>0$使得$U(x_0,\delta)\subseteq E$，并且当$0<||h||_X<\delta$时有：
	\begin{equation*}
		||R(h)||<\frac{c}{2}||h||_X,\;c>0
	\end{equation*}
	任取$h\in X$满足$0<||h||_X<\delta$，令$u=h/||h||_X$并考察函数：
	\begin{equation*}
		\varphi(t)=f(x_0+tu),\quad0\leqslant t\leqslant||h||_X
	\end{equation*}
	对任意的$0<t\leqslant||h||_X$有：
	\begin{align*}
		&\varphi'(t)=\operatorname{D}f(x_0+tu)u=\operatorname{D}fx_0(u)+\operatorname{D}^2fx_0(tu,u)+R(tu)u \\
		=&\operatorname{D}fx_0(u)+t\operatorname{D}^2fx_0(u,u)+R(tu)u\geqslant0+tc||u||_X^2+R(tu)u=tc+R(tu)u
	\end{align*}
	由\info{算子范数}可得：
	\begin{equation*}
		R(tu)u>-|R(tu)u|\geqslant-||R(tu)||\;||u||_X=-||R(tu)||
	\end{equation*}
	因为$||tu||=|t|\in(0,||h||_X]\subseteq(0,\delta)$，所以：
	\begin{equation*}
		||R(tu)||<\frac{c}{2}||tu||_X=\frac{ct}{2}
	\end{equation*}
	于是：
	\begin{equation*}
		\varphi'(t)\geqslant tc+R(tu)u\geqslant tc-||R(tu)||>\frac{tc}{2}>0
	\end{equation*}
	由\cref{theo:LagrangeMeanValueTheorem}可知：
	\begin{equation*}
		f(x_0+h)-f(x_0)=\varphi(||h||_X)-\varphi(0)>0
	\end{equation*}
	因此$x_0$是$f$的严格极小值点。
\end{proof}
\begin{lemma}
	设$A\in\mathbb{R}^{m\times n}$，则$[\operatorname{Ker}(A)]^\perp=\operatorname{Im}(A^{\top})$。
\end{lemma}
\begin{proof}
	任取$y\in\operatorname{Im}(A^{\top})$，则存在$z\in\mathbb{R}^m$使得$y=A^{\top}z$。因为对于任意$x\in\operatorname{Ker}(A)$有$Ax=\mathbf{0}$，于是由\cref{prop:Transpose}(4)可得：
	\begin{equation*}
		y^{\top}x=z^{\top}Ax=0
	\end{equation*}
	所以$y\in[\operatorname{Ker}(A)]^\perp$，根据$y$的任意性可得$\operatorname{Im}(A^{\top})\subseteq[\operatorname{Ker}(A)]^\perp$。\par
	由\cref{theo:SolutionOfSLE2}(3)可知：
	\begin{equation*}
		\dim\operatorname{Ker}(A)+\operatorname{rank}(A)=n
	\end{equation*}
	根据\cref{theo:OrthogonalComplementDirectSum}和\cref{theo:DirectSum}(4)可得：
	\begin{align*}
		\dim\{[\operatorname{Ker}(A)]^\perp\}=n-\dim\operatorname{Ker}(A)=\operatorname{rank}(A)
	\end{align*}
	由\cref{theo:SolutionOfSLE2}(3)的证明过程可得：
	\begin{equation*}
		\dim\operatorname{Im}(A^{\top})=\operatorname{rank}(A^{\top})=\operatorname{rank}(A)
	\end{equation*}
	所以$\dim\{[\operatorname{Ker}(A)]^\perp\}=\dim\operatorname{Im}(A^{\top})$。由\cref{prop:nDimensionalLinearSpace}(2)即可得出结论。
\end{proof}
\subsubsection{梯度}
\begin{definition}
	设$X$是实Hilbert空间，$E\subseteq X$是开集，函数$f:E\to\mathbb{R}$在$x_0\in E$处G\^ateaux可微，于是存在有界线性算子$\operatorname{D}_Gfx_0\in\mathcal{L}(X,\mathbb{R})$。由\cref{theo:LinearOperatorBoundedContinuous}(1)可得$\operatorname{D}_Gfx_0$是连续线性映射，根据\cref{theo:RieszRepresentation}可知存在唯一的$\nabla fx_0\in X$使得：
	\begin{equation*}
		\forall\;h\in X,\quad(\operatorname{D}_Gfx_0)h=(\nabla fx_0,h)
	\end{equation*}
	称$\nabla fx_0$为$f$在$x_0$处的\gls{Gradient}。
\end{definition}
\begin{note}
	当$X=\mathbb{R}^n$并取标准内积和标准基时，根据偏导数的定义、\cref{prop:PartialDifferential}(2)和梯度的定义可得：
	\begin{equation*}
		\partial_ifx_0=\operatorname{D}_{G,i}fx_0(1)=(\operatorname{D}_Gfx_0)\circ\iota_i(1)=\operatorname{D}_Gfx_0(e_i)=(\nabla fx_0,e_i)
	\end{equation*}
	设：
	\begin{equation*}
		\nabla fx_0=(\seq{a}{n})^{\top}
	\end{equation*}
	则$(\nabla fx_0,e_i)=a_i$，即：
	\begin{equation*}
		\nabla fx_0=(\partial_1fx_0,\partial_2fx_0,\dots,\partial_nfx_0)^{\top}
	\end{equation*}
	因此：
	\begin{equation*}
		\operatorname{D}_Gfx_0(h)=(\nabla fx_0)^{\top}h=\sum_{i=1}^{n}\partial_ifx_0h_i
	\end{equation*}
	若$f$在$x_0$处Fr\'echet可微，则由\cref{prop:FrechetGateaux}(1)可将上式中的$\operatorname{D}_Gfx_0$替换为$\operatorname{D}fx_0$。
\end{note}
\begin{property}\label{prop:Gradient}
	设$X$是实Hilbert空间，$E\subseteq X$是开集，函数$f:E\to\mathbb{R}$在$x_0\in E$处G\^ateaux可微。若$\nabla fx_0\ne\mathbf{0}$，则在所有单位方向$h\in X$中：
	\begin{equation*}
		\max_{||h||=1}\dif f(x_0;h)=||\nabla fx_0||,\quad
		\min_{||h||=1}\dif f(x_0;h)=-||\nabla fx_0||
	\end{equation*}
	最大值和最小值分别在：
	\begin{equation*}
		h=\frac{\nabla fx_0}{||\nabla fx_0||},\quad
		h=-\frac{\nabla fx_0}{||\nabla fx_0||}
	\end{equation*}
	处取到。若$\nabla fx_0=\mathbf{0}$，则$f$在$x_0$处沿任意单位方向的G\^ateaux方向导数都为$0$。
\end{property}
\begin{proof}
	由\cref{ineq:CauchySchiwarz}可知对于任意的单位方向$h\in X$有：
	\begin{equation*}
		|\dif f(x_0;h)|=|(\nabla fx_0,h)|
		\leqslant||\nabla fx_0||\,||h||=||\nabla fx_0||
	\end{equation*}
	当$\nabla fx_0\ne\mathbf{0}$时，将上述两个单位方向代入即可分别得到$||\nabla fx_0||$和$-||\nabla fx_0||$。当$\nabla fx_0=\mathbf{0}$时，由梯度的定义立即可得结论。
\end{proof}
